# Cursor-Tracking Character Hero Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full-viewport hero section where a pre-recorded character portrait's gaze follows the visitor's cursor in real time, using pre-extracted WebP frames drawn on a `<canvas>` — zero in-browser video, zero 3D transforms, zero frame ghosting.

**Architecture:** A Vite + React + TypeScript SPA. All per-frame math (angle calculation, circular lerp, cover-fit transform, frame-index selection) lives in small pure functions in `src/lib/`, unit-tested with Vitest. A single shared ticker (`src/lib/ticker.ts`) drives one `requestAnimationFrame` loop that both `CharacterCanvas` (head tracking) and `MagneticCursor` (custom cursor) subscribe to, reading cursor position from one shared mutable `interactionState` object — avoiding React re-renders and duplicate 60fps timers. Visual/integration pieces (canvas rendering, layout, responsiveness) are verified manually in a real browser via Playwright, not unit-tested.

**Tech Stack:** Vite 6, React 19, TypeScript 5, Vitest + jsdom + @testing-library/react for unit tests. No routing library, no CSS framework — hand-written CSS with CSS custom properties. Google Fonts (Dancing Script, Inter) loaded via `<link>`.

**Spec:** `docs/superpowers/specs/2026-09-25-cursor-tracking-hero-design.md`

## Global Constraints

- No CSS 3D transforms (`perspective`, `rotateX`, `rotateY`) anywhere on the page or canvas; only which bitmap is drawn on the canvas ever changes.
- Never seek or `play()` `public/character.mp4` in the browser. The 65 pre-extracted WebP frames in `public/frames/` (already generated by `scripts/extract_frames.py`) are the only runtime assets.
- Exactly one `drawImage` call per render tick, full opacity, no alpha-blending between frames.
- Background color everywhere (body, canvas) is `#64bce2`.
- Deadzone: cursor within `0.12 * Math.min(viewportWidth, viewportHeight)` of the face anchor shows `center.webp` regardless of angle.
- Angular smoothing factor: `0.26` per tick, shortest-path circular lerp.
- Frame convention: `frame_XX.webp` = true screen angle `XX * 5.625°`, where `atan2(dy, dx)` in y-down screen space gives `0°=RIGHT, 90°=DOWN, 180°=LEFT, 270°=UP`, increasing clockwise.
- Face anchor (the `atan2` origin) is normalized `(0.49, 0.38)` in the source video's native `1280×720` space, remapped through the live `object-fit: cover` transform.
- `Resume` button opens `public/resume.pdf` in a new tab; `Let's Talk` button is `mailto:sakshihaste1008@gmail.com`.
- `prefers-reduced-motion`: canvas freezes on `center.webp`, cursor dot/ring track the pointer with no trailing lag (no content ever hidden).
- Custom cursor (`cursor: none` + dot/ring) is disabled entirely on coarse/touch pointers.

// src/lib/angle.ts

/** Wraps any degree value into the range [0, 360). */
export function normalizeAngle(deg: number): number {
  const mod = deg % 360;
  return mod < 0 ? mod + 360 : mod;
}

/**
 * Angle in degrees from (originX, originY) to (pointX, pointY), using
 * atan2(dy, dx) in y-down screen space: 0=RIGHT, 90=DOWN, 180=LEFT, 270=UP.
 */
export function angleTo(
  originX: number,
  originY: number,
  pointX: number,
  pointY: number
): number {
  const dx = pointX - originX;
  const dy = pointY - originY;
  const rad = Math.atan2(dy, dx);
  return normalizeAngle((rad * 180) / Math.PI);
}

/** Shortest-path circular interpolation from `current` toward `target`. */
export function lerpAngle(current: number, target: number, factor: number): number {
  let delta = normalizeAngle(target - current);
  if (delta > 180) delta -= 360;
  return normalizeAngle(current + delta * factor);
}

export const FRAME_COUNT = 64;
export const DEGREES_PER_FRAME = 360 / FRAME_COUNT;

/** Nearest of the 64 pre-extracted rotation frames for a given angle. */
export function angleToFrameIndex(angleDeg: number): number {
  const normalized = normalizeAngle(angleDeg);
  return Math.round(normalized / DEGREES_PER_FRAME) % FRAME_COUNT;
}
```

// src/lib/coverTransform.ts

export interface CoverTransform {
  scale: number;
  offsetX: number;
  offsetY: number;
}

/**
 * Replicates CSS `object-fit: cover` math: the source is scaled up by
 * whichever axis needs more growth to fill the container, then centered,
 * so the container is always fully covered with the excess cropped evenly.
 */
export function computeCoverTransform(
  containerWidth: number,
  containerHeight: number,
  sourceWidth: number,
  sourceHeight: number
): CoverTransform {
  const scale = Math.max(containerWidth / sourceWidth, containerHeight / sourceHeight);
  const drawWidth = sourceWidth * scale;
  const drawHeight = sourceHeight * scale;
  return {
    scale,
    offsetX: (containerWidth - drawWidth) / 2,
    offsetY: (containerHeight - drawHeight) / 2,
  };
}

/**
 * Maps a point given as normalized source-space coordinates (0-1, 0-1)
 * through a cover transform into live container/viewport pixel coordinates.
 */
export function mapSourcePointToContainer(
  normX: number,
  normY: number,
  sourceWidth: number,
  sourceHeight: number,
  transform: CoverTransform
): { x: number; y: number } {
  const sourceX = normX * sourceWidth;
  const sourceY = normY * sourceHeight;
  return {
    x: sourceX * transform.scale + transform.offsetX,
    y: sourceY * transform.scale + transform.offsetY,
  };
}
```

// src/lib/trackingStep.ts
import { angleTo, angleToFrameIndex, lerpAngle } from "./angle";

export const LERP_FACTOR = 0.26;
export const DEADZONE_FRACTION = 0.12;

export interface TrackingStepInput {
  cursorX: number;
  cursorY: number;
  faceX: number;
  faceY: number;
  currentSmoothedAngle: number;
  viewportWidth: number;
  viewportHeight: number;
}

export interface TrackingStepResult {
  smoothedAngle: number;
  frameIndex: number;
  isCenter: boolean;
}

export function computeTrackingStep(input: TrackingStepInput): TrackingStepResult {
  const {
    cursorX,
    cursorY,
    faceX,
    faceY,
    currentSmoothedAngle,
    viewportWidth,
    viewportHeight,
  } = input;

  const dx = cursorX - faceX;
  const dy = cursorY - faceY;
  const distance = Math.hypot(dx, dy);
  const deadzoneRadius = DEADZONE_FRACTION * Math.min(viewportWidth, viewportHeight);
  const isCenter = distance < deadzoneRadius;

  const targetAngle = angleTo(faceX, faceY, cursorX, cursorY);
  const smoothedAngle = lerpAngle(currentSmoothedAngle, targetAngle, LERP_FACTOR);
  const frameIndex = angleToFrameIndex(smoothedAngle);

  return { smoothedAngle, frameIndex, isCenter };
}
```

// src/lib/frameAssets.ts

export const FRAME_COUNT = 64;
export const CENTER_FRAME_PATH = "/frames/center.webp";

/** Path for rotation frame `index`, wrapping into the valid 0-63 range. */
export function framePath(index: number): string {
  const wrapped = ((index % FRAME_COUNT) + FRAME_COUNT) % FRAME_COUNT;
  return `/frames/frame_${String(wrapped).padStart(2, "0")}.webp`;
}

export interface FrameSet {
  rotation: HTMLImageElement[];
  center: HTMLImageElement;
}

function loadImage(src: string): Promise<HTMLImageElement> {
  return new Promise((resolve, reject) => {
    const img = new Image();
    img.onload = () => resolve(img);
    img.onerror = () => reject(new Error(`failed to load ${src}`));
    img.src = src;
  });
}

/** Preloads all 64 rotation frames plus the dedicated center frame. */
export async function loadFrameSet(): Promise<FrameSet> {
  const rotationPromises = Array.from({ length: FRAME_COUNT }, (_, i) =>
    loadImage(framePath(i))
  );
  const [rotation, center] = await Promise.all([
    Promise.all(rotationPromises),
    loadImage(CENTER_FRAME_PATH),
  ]);
  return { rotation, center };
}

// src/lib/ticker.ts

type TickCallback = (timestamp: number) => void;

const subscribers = new Set<TickCallback>();
let rafId: number | null = null;

function loop(timestamp: number) {
  subscribers.forEach((callback) => callback(timestamp));
  if (subscribers.size > 0) {
    rafId = requestAnimationFrame(loop);
  } else {
    rafId = null;
  }
}

/**
 * Subscribes to a single shared requestAnimationFrame loop (rather than
 * each caller starting its own rAF timer) so CharacterCanvas and
 * MagneticCursor stay in lockstep on one 60fps tick. Returns an
 * unsubscribe function; the underlying rAF loop stops once the last
 * subscriber unsubscribes and restarts on the next new subscription.
 */
export function subscribeTick(callback: TickCallback): () => void {
  subscribers.add(callback);
  if (rafId === null) {
    rafId = requestAnimationFrame(loop);
  }
  return () => {
    subscribers.delete(callback);
    if (subscribers.size === 0 && rafId !== null) {
      cancelAnimationFrame(rafId);
      rafId = null;
    }
  };
}

// src/hooks/useInteractionTracking.ts
import { useEffect } from "react";
import { interactionState } from "../lib/interactionState";

const HOVER_ATTR = "data-cursor-hover";

/**
 * Wires window/document-level pointer listeners into the shared
 * interactionState singleton. Call this exactly once (in Hero.tsx).
 */
export function useInteractionTracking(): void {
  useEffect(() => {
    function handleMove(event: PointerEvent) {
      interactionState.x = event.clientX;
      interactionState.y = event.clientY;
    }
    function handleOver(event: PointerEvent) {
      const target = event.target as HTMLElement | null;
      if (target?.closest(`[${HOVER_ATTR}]`)) {
        interactionState.hovering = true;
      }
    }
    function handleOut(event: PointerEvent) {
      const related = event.relatedTarget as HTMLElement | null;
      if (!related || !related.closest(`[${HOVER_ATTR}]`)) {
        interactionState.hovering = false;
      }
    }

    window.addEventListener("pointermove", handleMove);
    document.addEventListener("pointerover", handleOver);
    document.addEventListener("pointerout", handleOut);
    return () => {
      window.removeEventListener("pointermove", handleMove);
      document.removeEventListener("pointerover", handleOver);
      document.removeEventListener("pointerout", handleOut);
    };
  }, []);
}
```

// src/hooks/useReducedMotion.ts
import { useEffect, useState } from "react";

const QUERY = "(prefers-reduced-motion: reduce)";

export function useReducedMotion(): boolean {
  const [reduced, setReduced] = useState<boolean>(() => window.matchMedia(QUERY).matches);

  useEffect(() => {
    const mql = window.matchMedia(QUERY);
    const handler = (event: MediaQueryListEvent) => setReduced(event.matches);
    mql.addEventListener("change", handler);
    return () => mql.removeEventListener("change", handler);
  }, []);

  return reduced;
}

```tsx
// src/components/CharacterCanvas.tsx
import { useEffect, useRef } from "react";
import { computeCoverTransform, mapSourcePointToContainer } from "../lib/coverTransform";
import { loadFrameSet, type FrameSet } from "../lib/frameAssets";
import { interactionState } from "../lib/interactionState";
import { subscribeTick } from "../lib/ticker";
import { computeTrackingStep } from "../lib/trackingStep";
import { useReducedMotion } from "../hooks/useReducedMotion";

const SOURCE_WIDTH = 1280;
const SOURCE_HEIGHT = 720;
const FACE_ANCHOR_NORM = { x: 0.49, y: 0.38 };

export function CharacterCanvas() {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const smoothedAngleRef = useRef(0);
  const framesRef = useRef<FrameSet | null>(null);
  const reducedMotion = useReducedMotion();

  useEffect(() => {
    let cancelled = false;
    loadFrameSet().then((frames) => {
      if (!cancelled) framesRef.current = frames;
    });
    return () => {
      cancelled = true;
    };
  }, []);

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas?.getContext("2d");
    if (!canvas || !ctx) return;

    function resize() {
      const dpr = window.devicePixelRatio || 1;
      canvas!.width = window.innerWidth * dpr;
      canvas!.height = window.innerHeight * dpr;
      ctx!.setTransform(dpr, 0, 0, dpr, 0, 0);
    }
    resize();
    window.addEventListener("resize", resize);

    const unsubscribe = subscribeTick(() => {
      const frames = framesRef.current;
      if (!frames) return;

      const viewportWidth = window.innerWidth;
      const viewportHeight = window.innerHeight;
      const transform = computeCoverTransform(
        viewportWidth,
        viewportHeight,
        SOURCE_WIDTH,
        SOURCE_HEIGHT
      );
      const faceAnchor = mapSourcePointToContainer(
        FACE_ANCHOR_NORM.x,
        FACE_ANCHOR_NORM.y,
        SOURCE_WIDTH,
        SOURCE_HEIGHT,
        transform
      );

      let image: HTMLImageElement;
      if (reducedMotion) {
        image = frames.center;
      } else {
        const step = computeTrackingStep({
          cursorX: interactionState.x,
          cursorY: interactionState.y,
          faceX: faceAnchor.x,
          faceY: faceAnchor.y,
          currentSmoothedAngle: smoothedAngleRef.current,
          viewportWidth,
          viewportHeight,
        });
        smoothedAngleRef.current = step.smoothedAngle;
        image = step.isCenter ? frames.center : frames.rotation[step.frameIndex];
      }

      ctx.clearRect(0, 0, viewportWidth, viewportHeight);
      ctx.drawImage(
        image,
        transform.offsetX,
        transform.offsetY,
        SOURCE_WIDTH * transform.scale,
        SOURCE_HEIGHT * transform.scale
      );
    });

    return () => {
      window.removeEventListener("resize", resize);
      unsubscribe();
    };
  }, [reducedMotion]);

  return (
    <canvas
      ref={canvasRef}
      role="img"
      aria-label="Portrait of Sakshi Haste, whose gaze follows your cursor"
      style={{
        position: "fixed",
        inset: 0,
        display: "block",
        width: "100vw",
        height: "100vh",
        background: "#64bce2",
      }}
    />
  );
}
```

**Interfaces:**

- Produces: `MagneticCursor` component (named export).

- [ ] **Step 1: Write `src/components/MagneticCursor.tsx`**

No unit test: this is a real-time DOM-transform animation with no meaningful jsdom assertion (there is no layout/paint to inspect). Verified visually in Task 12.

```tsx
// src/components/MagneticCursor.tsx
import { useEffect, useRef } from "react";
import { interactionState } from "../lib/interactionState";
import { subscribeTick } from "../lib/ticker";
import { useReducedMotion } from "../hooks/useReducedMotion";

const DOT_LERP = 0.6;
const RING_LERP = 0.18;
const HOVER_SCALE = 1.8;

export function MagneticCursor() {
  const dotRef = useRef<HTMLDivElement | null>(null);
  const ringRef = useRef<HTMLDivElement | null>(null);
  const dotPos = useRef({ x: -100, y: -100 });
  const ringPos = useRef({ x: -100, y: -100 });
  const reducedMotion = useReducedMotion();
  const isCoarsePointerRef = useRef(false);

  useEffect(() => {
    isCoarsePointerRef.current = window.matchMedia("(pointer: coarse)").matches;
    if (isCoarsePointerRef.current) return;

    const unsubscribe = subscribeTick(() => {
      const targetX = interactionState.x;
      const targetY = interactionState.y;
      const dotFactor = reducedMotion ? 1 : DOT_LERP;
      const ringFactor = reducedMotion ? 1 : RING_LERP;

      dotPos.current.x += (targetX - dotPos.current.x) * dotFactor;
      dotPos.current.y += (targetY - dotPos.current.y) * dotFactor;
      ringPos.current.x += (targetX - ringPos.current.x) * ringFactor;
      ringPos.current.y += (targetY - ringPos.current.y) * ringFactor;

      const scale = interactionState.hovering ? HOVER_SCALE : 1;

      if (dotRef.current) {
        dotRef.current.style.transform = `translate3d(${dotPos.current.x}px, ${dotPos.current.y}px, 0) translate(-50%, -50%)`;
      }
      if (ringRef.current) {
        ringRef.current.style.transform = `translate3d(${ringPos.current.x}px, ${ringPos.current.y}px, 0) translate(-50%, -50%) scale(${scale})`;
      }
    });

    return unsubscribe;
  }, [reducedMotion]);

  if (typeof window !== "undefined" && window.matchMedia("(pointer: coarse)").matches) {
    return null;
  }

  return (
    <>
      <div ref={dotRef} className="magnetic-cursor-dot" aria-hidden="true" />
      <div ref={ringRef} className="magnetic-cursor-ring" aria-hidden="true" />
    </>
  );
}
```
